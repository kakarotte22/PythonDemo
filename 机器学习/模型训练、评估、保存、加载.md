# -*- coding: utf-8 -*-

# ============================================================================

# 第一部分: 导入依赖库

# ============================================================================

import os # 操作系统接口，用于读取环境变量、文件路径操作

import json # JSON 序列化/反序列化，用于保存模型元数据

import tempfile # 临时文件管理，用于模型格式转换时的中间存储

from datetime import datetime # 日期时间处理，用于生成时间戳

from itertools import product # 笛卡尔积，用于生成网格搜索的参数组合

  

# ----------------------------------------------------------------------------

# PySpark 核心模块

# ----------------------------------------------------------------------------

from pyspark.sql import SparkSession # Spark 会话入口，所有 Spark 操作的起点

from pyspark.sql.functions import col # DataFrame 操作函数

from pyspark.ml.feature import VectorAssembler # 特征工程工具

from pyspark import StorageLevel # 缓存存储级别控制

from pyspark.ml.functions import vector_to_array

from pyspark.sql import functions as F

# ----------------------------------------------------------------------------

# XGBoost for PySpark - XGBoost 官方提供的 PySpark 集成

# 提供了 XGBoost 的分布式实现，支持在 Spark 集群上训练 XGBoost 模型

# ----------------------------------------------------------------------------

from xgboost.spark import SparkXGBClassifier

from xgboost import XGBClassifier

# ----------------------------------------------------------------------------

# 模型评估器 - 用于计算模型性能指标

# ----------------------------------------------------------------------------

from pyspark.ml.evaluation import (

BinaryClassificationEvaluator, # 二分类评估: AUC, PR曲线等

MulticlassClassificationEvaluator # 多分类评估: 准确率, F1等

)

  

# ============================================================================

# 第二部分: Spark Session 初始化

# ============================================================================

spark = SparkSession.builder \

.appName('XGB_Train') \

.config("spark.sql.execution.arrow.pyspark.enabled", "true") \

.config("spark.sql.adaptive.enabled", "true") \

.config("spark.sql.adaptive.coalescePartitions.enabled", "true") \

.config("spark.sql.shuffle.partitions", "2000") \

.getOrCreate()

  

# 配置 OSS 访问

# spark.sparkContext._jsc.hadoopConfiguration().set("fs.oss.accessKeyId", spark.conf.get("spark.accessKeyId"))

# spark.sparkContext._jsc.hadoopConfiguration().set("fs.oss.accessKeySecret", spark.conf.get("spark.accessKeySecret"))

# spark.sparkContext._jsc.hadoopConfiguration().set("fs.oss.endpoint", 'oss-cn-hangzhou-internal.aliyuncs.com')

  

print(f"Spark Version: {spark.version}")

print(f"Scala Version: {spark.sparkContext._jsc.sc().version()}")

  

# ============================================================================

# 第三部分: 载入数据

# ============================================================================

# 训练集数据

TRAIN_TABLE_NAME = spark.conf.get("spark.train_table_name")

train_df = spark.read.table(TRAIN_TABLE_NAME)

  

# 验证集数据

VAL_TABLE_NAME = spark.conf.get("spark.val_table_name")

val_df = spark.read.table(VAL_TABLE_NAME)

  

# 标签列名：训练集和验证集数据结构必须相同

LABEL_COL = spark.conf.get("spark.label_col")

  

# 需要保留但不参与训练的字段，逗号分隔，如 "request_id,user_id"

# 若未配置则默认为空

_keep_cols_str = spark.conf.get("spark.keep_cols", "")

_keep_cols_raw = [c.strip() for c in _keep_cols_str.split(",") if c.strip()]

  

# 过滤掉训练集中不存在的列，避免后续 select 报错

_train_col_set = set(train_df.columns)

_invalid_keep = [c for c in _keep_cols_raw if c not in _train_col_set]

if _invalid_keep:

print(f"警告: keep_cols 中以下字段在训练集中不存在，已自动剔除: {_invalid_keep}")

KEEP_COLS = [c for c in _keep_cols_raw if c in _train_col_set]

print(f"Keep Cols (不参与训练，打分结果保留): {KEEP_COLS}")

  

# ============================================================================

# 第四部分: 缺失值填充与向量化

# ============================================================================

# --- 标签比例采样，保证正样本(label=1)占比约为 10% ---

_label_counts = train_df.groupBy(LABEL_COL).count().collect()

_count_map = {row[LABEL_COL]: row["count"] for row in _label_counts}

_pos_count = _count_map.get(1, 0)

_neg_count = _count_map.get(0, 0)

_total_count = _pos_count + _neg_count

  

print(f"原始数据集: 总样本={_total_count}, 正样本(label=1)={_pos_count}, 负样本(label=0)={_neg_count}")

if _total_count == 0:

raise ValueError("训练集为空，请检查输入表！")

if _pos_count == 0:

raise ValueError("训练集中正样本数量为 0，无法训练分类模型！")

print(f"原始正样本占比: {_pos_count / _total_count * 100:.2f}%")

  

# 目标正样本占比 10%: pos / (pos + neg) = 0.1 => neg_needed = pos * 9

_TARGET_POS_RATIO = 0.10

_neg_needed = int(_pos_count * (1 - _TARGET_POS_RATIO) / _TARGET_POS_RATIO)

  

if _neg_needed >= _neg_count:

print(f"负样本数量({_neg_count})未超过目标({_neg_needed})，无需采样，直接使用全量数据")

else:

_neg_fraction = min(_neg_needed / _neg_count, 1.0)

print(f"负样本过多，进行随机采样: 采样比例={_neg_fraction:.6f}，采样后负样本数≈{_neg_needed}")

_df_pos = train_df.filter(col(LABEL_COL) == 1)

_df_neg = train_df.filter(col(LABEL_COL) == 0).sample(fraction=_neg_fraction, seed=42)

train_df = _df_pos.union(_df_neg)

_sampled_counts = train_df.groupBy(LABEL_COL).count().collect()

_sampled_map = {row[LABEL_COL]: row["count"] for row in _sampled_counts}

_s_pos = _sampled_map.get(1, 0)

_s_neg = _sampled_map.get(0, 0)

_s_total = _s_pos + _s_neg

print(f"采样后数据集: 总样本={_s_total}, 正样本={_s_pos}, 负样本={_s_neg}")

print(f"采样后正样本占比: {_s_pos / _s_total * 100:.2f}%")

  

# 排除标签列和 keep_cols，其余列作为特征

_exclude_cols = set([LABEL_COL] + KEEP_COLS)

feature_cols = [c for c in train_df.columns if c not in _exclude_cols]

  

# [优化] fillna 只作用于特征列，避免错误填充标签列/字符串列

train_df = train_df.fillna(-1, subset=feature_cols)

val_df = val_df.fillna(-1, subset=feature_cols)

  

featurizer = VectorAssembler(inputCols=feature_cols, outputCol="features")

  

# val_df 校验：确保验证集包含所有特征列，否则 VectorAssembler 会报 UNRESOLVED_COLUMN

_val_col_set = set(val_df.columns)

_missing_in_val = [c for c in feature_cols if c not in _val_col_set]

if _missing_in_val:

raise ValueError(f"验证集缺少以下特征列，无法进行 transform: {_missing_in_val}")

  

# [优化] transform 后立即 select 最小必要列集，丢弃原始特征宽列，减少缓存内存占用

_cache_cols = list(KEEP_COLS) + [LABEL_COL, "features"]

train_df = featurizer.transform(train_df).select(*_cache_cols)

val_df = featurizer.transform(val_df).select(*_cache_cols)

  

# [优化] 使用 MEMORY_AND_DISK_DESER 序列化缓存，比默认 cache() 内存占用更小

train_df.persist(StorageLevel.MEMORY_AND_DISK_DESER)

val_df.persist(StorageLevel.MEMORY_AND_DISK_DESER)

# 触发 action 让缓存物化，避免后续训练时重复计算

train_df.count()

val_df.count()

  

# ============================================================================

# 第五部分: 网格搜索

# ============================================================================

# 存储搜索结果的属性

best_model = None # 最佳模型 (训练好的模型对象)

best_params = None # 最佳参数字典

best_score = None # 最佳评分

cv_results = [] # 所有参数组合的结果列表

  
  

# 创建 XGBoost 专用的参数空间

def get_default_param_grid():

"""

XGBoost 参数网格

"""

return {

# 树结构参数

"max_depth": [3,5], # 树的最大深度

"max_leaves": [32], # 最大叶子节点数 (0表示不限制)

  

# 学习参数

"learning_rate": [0.1], # 学习率 (eta)

"n_estimators": [100, 300, 500], # 树的数量

  

# 防过拟合参数

"min_child_weight": [50,100], # 叶子节点最小样本权重和

"subsample": [0.7], # 样本采样比例

"colsample_bytree": [0.7], # 特征采样比例

  
  

# 正则化参数

"reg_alpha": [0.1], # L1 正则化

"reg_lambda": [0.1], # L2 正则化

  

# 额外的 XGBoost 特定参数

"gamma": [0], # 节点分裂最小损失减少值

}

  

# 生成所有超参数组合

def generate_param_combinations(param_grid):

"""

生成所有超参数组合

"""

keys = param_grid.keys()

values = param_grid.values()

  

# 使用 itertools.product 计算笛卡尔积

combinations = [dict(zip(keys, v)) for v in product(*values)]

  

print(f"\n Param Search Space:")

for key, vals in param_grid.items():

print(f" - {key}: {vals}")

print(f"\n Total param combinations Cnt: {len(combinations)}")

  

# 估算训练时间

if len(combinations) > 100:

print(f"警告: 参数组合数较多，建议减少参数候选值")

  

return combinations

  

# 评估模型性能

def evaluate_model(model, test_df, label_col="label"):

"""

评估模型性能

"""

predictions = model.transform(test_df)

# [优化] 缓存 predictions，两个 evaluator 复用同一份结果，避免重复触发 transform

predictions.cache()

  

try:

# 二分类评估

auc_evaluator = BinaryClassificationEvaluator(

labelCol=label_col,

rawPredictionCol="rawPrediction",

metricName="areaUnderROC"

)

auc = auc_evaluator.evaluate(predictions)

  

# 多分类评估（用于准确率）

acc_evaluator = MulticlassClassificationEvaluator(

labelCol=label_col,

predictionCol="prediction",

metricName="accuracy"

)

accuracy = acc_evaluator.evaluate(predictions)

finally:

predictions.unpersist()

  

return {

"auc": auc,

"accuracy": accuracy,

"primary_metric": auc # 使用 AUC 作为主要评估指标

}

  

# 网格搜索

def grid_search(train_df, valid_df, param_grid=None, label_col="label", features_col="features"):

"""

执行 XGBoost 网格搜索

"""

global best_model, best_params, best_score, cv_results

  

# 如果未指定参数网格，使用默认配置

if param_grid is None:

param_grid = get_default_param_grid()

  

# 生成所有参数组合

param_combinations = generate_param_combinations(param_grid)

  

# 初始化最佳分数为负无穷

best_score = float("-inf")

  

# 记录开始时间

start_time = datetime.now()

  

# ---------------------------------------------------------------------

# 遍历所有参数组合

# ---------------------------------------------------------------------

for i, params in enumerate(param_combinations):

print(f"\n{'='*60}")

print(f"Train View: {i+1}/{len(param_combinations)} " +

f"({(i+1)/len(param_combinations)*100:.1f}%)")

print(f"{'='*60}")

print(f"Current Param Combination:")

for k, v in params.items():

print(f" - {k}: {v}")

  

try:

# 步骤1: 创建 XGBoost 分类器

# SparkXGBClassifier 的基本参数设置

xgb_params = {

"features_col": features_col,

"label_col": label_col,

"prediction_col": "prediction",

"probability_col": "probability",

"raw_prediction_col": "rawPrediction",

"eval_metric": "auc", # 评估指标

"num_workers": 20, # 工作节点数

"use_gpu": False, # 是否使用GPU

"scale_pos_weight": 1, # 处理不平衡数据

"seed": 42, # 随机种子

}

  

# 合并网格搜索参数

xgb_params.update(params)

  

# 创建模型

model = SparkXGBClassifier(**xgb_params)

print('创建 XGBoost 模型成功！')

  

# 步骤2: 训练模型

iter_start = datetime.now()

trained_model = model.fit(train_df)

train_time = (datetime.now() - iter_start).total_seconds()

print(f"\n 训练耗时: {train_time:.2f} 秒")

  

# 步骤3: 评估模型

metrics = evaluate_model(trained_model, valid_df, label_col)

score = metrics["primary_metric"]

  

print(f"评估结果:")

for k, v in metrics.items():

if k != "primary_metric":

print(f" - {k}: {v:.4f}")

  

# 步骤4: 记录结果

cv_results.append({

"params": params,

"metrics": metrics,

"score": score,

"train_time": train_time

})

  

# 步骤5: 更新最佳模型

if score > best_score:

best_score = score

best_params = params

best_model = trained_model

print(f"\n 发现更好的模型!")

print(f" 当前最佳分数: {score:.4f}")

else:

# [优化] 非最佳模型立即释放引用，减少 Driver 内存压力

del trained_model

  

except Exception as e:

print(f"训练失败: {e}")

cv_results.append({

"params": params,

"error": str(e)

})

  

# ---------------------------------------------------------------------

# 输出最终结果

# ---------------------------------------------------------------------

total_time = (datetime.now() - start_time).total_seconds()

  

print(f"\n{'='*60}")

print(f" Grid Search 完成!")

print(f"{'='*60}")

print(f"总耗时: {total_time/60:.2f} 分钟")

print(f"参数组合总数: {len(param_combinations)}")

print(f"成功训练次数: {len([r for r in cv_results if r.get('score') is not None])}")

print(f"\n 最佳参数:")

for k, v in best_params.items():

print(f" - {k}: {v}")

print(f"\n 最佳分数: {best_score:.4f}")

  

return best_model, best_params, cv_results

  
  

# 执行网格搜索

best_model, best_params, cv_results = grid_search(train_df, val_df, label_col=LABEL_COL)

raise RuntimeError("所有参数组合均训练失败，best_model 为 None，无法继续执行打分。请检查以上训练错误日志。")

print(f"Best Score: {best_score:.4f}")

print(f"CV Results Count: {len(cv_results)}")

print(f"Best Model: {best_model}, type: {type(best_model)}")

def score_and_save(model, df, table_name, label_col, feature_cols, keep_cols, dataset_name="dataset"):

"""

对数据集打分，提取 probability_1，并将结果保存到表。

保存列：keep_cols + label + probability_1

注意：df 已经过 VectorAssembler.select 处理，原始特征列已不存在，无需再 select feature_cols。

"""

print(f"\n{'='*60}")

print(f" 对 {dataset_name} 打分，保存到: {table_name}")

print(f"{'='*60}")

  

# [修复] df 经过 featurizer.transform().select(_cache_cols) 后只保留了

# keep_cols + label_col + features，原始特征列已丢弃，不可再 select。

# 去重保序，避免 label_col 同时出现在 keep_cols 中导致列名重复

_seen = set()

cols_keep = []

for c in list(keep_cols) + [label_col]:

if c not in _seen:

_seen.add(c)

cols_keep.append(c)

df_result = (

model.transform(df)

.select(

*cols_keep,

vector_to_array(col("probability"))[1].alias("probability_1")

)

)

  

df_result.write.mode("overwrite").saveAsTable(table_name)

print(f" {dataset_name} 打分结果已保存到: {table_name}")

return df_result

  
  

# 生成打分表名

TASK_ID = spark.conf.get("spark.task_id")

  

train_scores_table = 'du_risk_training.model_train_skills_predictions_train_' + TASK_ID

val_scores_table = 'du_risk_training.model_train_skills_predictions_val_' + TASK_ID

print(f"训练集打分表: {train_scores_table}")

print(f"验证集打分表: {val_scores_table}")

# 第七部分: 评估报告（等距20分箱）

# ============================================================================

def generate_eval_report(scores_df, label_col, score_col="probability_1", n_bins=20, dataset_name="dataset"):

"""

将模型评分按等距划分成 n_bins 份，统计每份内：

- 总样本数

- 占总样本数的比例

- 坏样本数（label=1）

- 坏样本占比（坏样本数 / 该箱总样本数）

- 坏样本占总坏样本的比例

返回 Spark DataFrame，同时打印到控制台。

"""

print(f"\n{'='*60}")

print(f" {dataset_name} 评估报告（等距 {n_bins} 分箱）")

print(f"{'='*60}")

  

# [优化] 一次 agg 同时计算总样本数和总坏样本数，避免两次全量扫描

stats = scores_df.agg(

F.count("*").alias("total_cnt"),

F.sum(F.when(F.col(label_col) == 1, 1).otherwise(0)).alias("total_bad")

).first()

total_cnt = stats["total_cnt"]

total_bad = stats["total_bad"]

  

print(f"总样本数: {total_cnt}, 总坏样本数: {total_bad}")

  

# 等距分箱：将 [0, 1] 分成 n_bins 份，箱号从 1 开始

bin_width = 1.0 / n_bins

scored = scores_df.withColumn(

"bin",

F.least(

F.lit(n_bins),

(F.floor(F.col(score_col) / F.lit(bin_width)) + 1).cast("int")

)

)

  

# 按箱聚合

report = (

scored.groupBy("bin")

.agg(

F.count("*").alias("total_cnt"),

F.sum(F.when(F.col(label_col) == 1, 1).otherwise(0)).alias("bad_cnt")

)

.withColumn("score_range",

F.concat(

F.lit("["),

F.round((F.col("bin") - 1) * F.lit(bin_width), 4).cast("string"),

F.lit(", "),

F.round(F.col("bin") * F.lit(bin_width), 4).cast("string"),

F.lit(")")

)

)

.withColumn("total_ratio", F.round(F.col("total_cnt") / F.lit(total_cnt), 6))

.withColumn("bad_rate", F.round(F.col("bad_cnt") / F.col("total_cnt"), 6))

.withColumn("bad_pct_total", F.round(F.col("bad_cnt") / F.lit(total_bad), 6))

.orderBy("bin")

.select(

F.col("bin"),

F.col("score_range"),

F.col("total_cnt"),

F.col("total_ratio"),

F.col("bad_cnt"),

F.col("bad_rate"),

F.col("bad_pct_total")

)

)

  

report.show(n_bins, truncate=False)

return report

  
  

train_report_table = 'du_risk_training.model_train_skills_predictions_train_report_' + TASK_ID

val_report_table = 'du_risk_training.model_train_skills_predictions_val_report_' + TASK_ID

  

train_report_df = generate_eval_report(train_scores_df, LABEL_COL, dataset_name="训练集")

train_report_df.write.mode("overwrite").saveAsTable(train_report_table)

print(f"训练集评估报告已保存到: {train_report_table}")

  

val_report_df = generate_eval_report(val_scores_df, LABEL_COL, dataset_name="验证集")

val_report_df.write.mode("overwrite").saveAsTable(val_report_table)

print(f"验证集评估报告已保存到: {val_report_table}")

def save_booster_to_table(model, feature_cols, table_name):

"""

将最佳模型的 booster 提取为 JSON 格式内容，

连同特征名称列表一起保存到 Spark 表，表结构：

- feature_names : string (逗号拼接的特征名列表)

- booster_json : string (booster 的 JSON 字符串内容)

"""

print(f"\n{'='*60}")

print(f" 保存 Booster 到表: {table_name}")

print(f"{'='*60}")

  

booster = model.get_booster()

  

# 将 booster 保存为临时 JSON 文件，再读取内容

with tempfile.NamedTemporaryFile(mode='w', suffix='.json', delete=False) as tmp:

tmp_path = tmp.name

  

booster.save_model(tmp_path)

with open(tmp_path, 'r', encoding='utf-8') as f:

booster_json_str = f.read()

os.remove(tmp_path)

  

feature_names_str = ",".join(feature_cols)

  

# 构造单行 DataFrame 存入表

booster_df = spark.createDataFrame(

[(feature_names_str, booster_json_str)],

schema=["feature_names", "booster_json"]

)

  

booster_df.write.mode("overwrite").saveAsTable(table_name)

print(f" Booster 已保存到: {table_name}")

print(f" 特征数量: {len(feature_cols)}")

  

# 顺便打印特征重要性

feature_importances = model.get_feature_importances(importance_type="gain")

print("Feature Importances (Gain):", feature_importances)

  
  

# 生成 booster 表名（以训练集表名为基准）

booster_table = 'du_risk_training.model_train_skills_predictions_train_booster_' + TASK_ID

save_booster_to_table(best_model, feature_cols, booster_table)

print(f"Booster 已保存到: {booster_table}")

# 第九部分: 从表加载模型并验证

# ============================================================================

def load_booster_from_table(table_name):

"""

从 Spark 表中读取 booster_json 和 feature_names，

还原为 XGBClassifier 对象，并打印模型基本信息以验证加载正确性。

返回: (loaded_model: XGBClassifier, loaded_feature_cols: list)
